Experiment 1 – Widgets and Layouts

Aim: Use Container, Row, Column, Expanded, Card and ListView.

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(title: const Text('Student Dashboard')),
        body: Column(
          children: [
            Container(
              padding: const EdgeInsets.all(20),
              child: const Text('Student Profile'),
            ),
            Row(
              mainAxisAlignment: MainAxisAlignment.spaceEvenly,
              children: const [
                Icon(Icons.home),
                Icon(Icons.person),
                Icon(Icons.settings),
              ],
            ),
            Expanded(
              child: ListView(
                children: const [
                  Card(child: ListTile(title: Text('Flutter'))),
                  Card(child: ListTile(title: Text('Cloud Computing'))),
                  Card(child: ListTile(title: Text('Programming'))),
                  Card(child: ListTile(title: Text('AI'))),
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }
}
Remember: Container, Row, Column, Expanded, ListView, Card.

Experiment 2 – Responsive UI Design

Aim: Make the layout change according to screen width using LayoutBuilder.

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(home: ResponsivePage());
  }
}

class ResponsivePage extends StatelessWidget {
  const ResponsivePage({super.key});

  Widget myCard(String name) {
    return Card(
      child: ListTile(
        leading: const Icon(Icons.school),
        title: Text(name),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Responsive Dashboard')),
      body: LayoutBuilder(
        builder: (context, constraints) {
          if (constraints.maxWidth < 600) {
            return SingleChildScrollView(
              child: Column(
                children: [
                  myCard('Student'),
                  myCard('Courses'),
                  myCard('Attendance'),
                  myCard('Results'),
                ],
              ),
            );
          } else {
            return Column(
              children: [
                Row(children: [
                  Expanded(child: myCard('Student')),
                  Expanded(child: myCard('Courses')),
                ]),
                Row(children: [
                  Expanded(child: myCard('Attendance')),
                  Expanded(child: myCard('Results')),
                ]),
              ],
            );
          }
        },
      ),
    );
  }
}
Remember: Mobile = Column. Wide screen = Row + Expanded. LayoutBuilder checks screen width.

Experiment 3 – Input and Forms

Aim: Create a simple student form for name and email.

import 'package:flutter/material.dart';

void main() {
  runApp(MaterialApp(home: MyApp()));
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Student Form')),
      body: Padding(
        padding: EdgeInsets.all(20),
        child: Column(
          children: [
            TextField(
              decoration: InputDecoration(labelText: 'Name'),
            ),
            TextField(
              decoration: InputDecoration(labelText: 'Email'),
            ),
            ElevatedButton(
              onPressed: () {
                print('Form Submitted');
              },
              child: Text('Submit'),
            ),
          ],
        ),
      ),
    );
  }
}
Remember: TextField + TextField + ElevatedButton. TextField accepts input; onPressed runs when tapped.

This is the simplified input version; it does not include Form validation.

Experiment 4 – Assets and Media

Aim: Display images from the assets folder.

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(title: const Text('Media Gallery')),
        body: GridView.count(
          crossAxisCount: 2,
          children: [
            Image.asset('assets/images/flutter.png'),
            Image.asset('assets/images/flutter.png'),
            Image.asset('assets/images/flutter.png'),
            Image.asset('assets/images/flutter.png'),
          ],
        ),
      ),
    );
  }
}

Folder structure

assets/
  images/
    flutter.png
lib/
  main.dart
pubspec.yaml

In pubspec.yaml, under flutter:

flutter:
  uses-material-design: true
  assets:
    - assets/images/
Remember: GridView.count makes a grid. Image.asset loads an image. Run flutter pub get after editing pubspec.yaml.

Experiment 5 – Navigation and Routing

Aim: Move between two screens using Navigator.

import 'package:flutter/material.dart';

void main() {
  runApp(MaterialApp(home: HomePage()));
}

class HomePage extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Home')),
      body: Center(
        child: ElevatedButton(
          onPressed: () {
            Navigator.push(
              context,
              MaterialPageRoute(
                builder: (context) => SecondPage(),
              ),
            );
          },
          child: Text('Go to Second Page'),
        ),
      ),
    );
  }
}

class SecondPage extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Second Page')),
      body: Center(
        child: ElevatedButton(
          onPressed: () {
            Navigator.pop(context);
          },
          child: Text('Go Back'),
        ),
      ),
    );
  }
}
Remember: Navigator.push() = open next screen. Navigator.pop() = go back.

Publish the website with GitHub Pages

  1. Download this HTML file and rename it to index.html.
  2. Sign in at github.com.
  3. Create a new repository named Flutter and set it to Public.
  4. Open the repository → Add file → Upload files.
  5. Upload index.html, then click Commit changes.
  6. Go to Settings → Pages.
  7. Under Build and deployment, select Deploy from a branch.
  8. Choose branch main and folder /(root), then click Save.
  9. After publishing, the address will look like https://YOUR-USERNAME.github.io/Flutter/.
Already have the Flutter repository? Upload this new file as index.html and replace the old one. Keep your existing Pages settings; your site address should stay the same.

To update later: Upload the updated index.html, replace the old file, and commit the changes.
